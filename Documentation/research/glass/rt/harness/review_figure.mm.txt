// Builds the review figure: 2x2 panels of the plugin's raw output texture,
// flipped upright (kernel row 0 = bottom of the view), brightened, with the
// true pane footprint (analytic, Unity projection) outlined in green.
#import <Foundation/Foundation.h>
#import <CoreGraphics/CoreGraphics.h>
#import <CoreText/CoreText.h>
#import <ImageIO/ImageIO.h>
#import <UniformTypeIdentifiers/UniformTypeIdentifiers.h>
#include <vector>
#include <string>
#include <algorithm>

static CGImageRef Load(const char* path)
{
    NSURL* url = [NSURL fileURLWithPath:[NSString stringWithUTF8String:path]];
    CGImageSourceRef src = CGImageSourceCreateWithURL((__bridge CFURLRef)url, nullptr);
    CGImageRef img = CGImageSourceCreateImageAtIndex(src, 0, nullptr);
    CFRelease(src);
    return img;
}

static void Text(CGContextRef ctx, const char* s, double x, double y, double size, CGColorRef color)
{
    CTFontRef font = CTFontCreateWithName(CFSTR("Helvetica-Bold"), size, nullptr);
    NSDictionary* attrs = @{(id)kCTFontAttributeName: (__bridge id)font, (id)kCTForegroundColorAttributeName: (__bridge id)color};
    NSAttributedString* as = [[NSAttributedString alloc] initWithString:[NSString stringWithUTF8String:s] attributes:attrs];
    CTLineRef line = CTLineCreateWithAttributedString((__bridge CFAttributedStringRef)as);
    CGContextSetTextPosition(ctx, x, y);
    CTLineDraw(line, ctx);
    CFRelease(line); CFRelease(font);
}

int main(int argc, char** argv)
{
    @autoreleasepool {
        // argv: out.jpg, then 4 x (png, title)
        const int SW = 1280, SH = 720;            // source size
        const double s = 770.0 / SW;              // panel scale
        const int PW = 770, PH = (int)(SH * s), GAP = 20, HEAD = 46;
        const int W = 2 * PW + 3 * GAP, H = 2 * (PH + HEAD) + 3 * GAP;
        CGColorSpaceRef cs = CGColorSpaceCreateDeviceRGB();
        CGContextRef ctx = CGBitmapContextCreate(nullptr, W, H, 8, W * 4, cs, kCGImageAlphaPremultipliedLast);
        CGContextSetRGBFillColor(ctx, .08, .08, .09, 1); CGContextFillRect(ctx, CGRectMake(0, 0, W, H));
        CGColorRef white = CGColorCreateGenericRGB(.95, .95, .95, 1), grey = CGColorCreateGenericRGB(.7, .7, .72, 1);
        for (int i = 0; i < 4; i++) {
            const char* png = argv[2 + i * 2];
            const char* title = argv[3 + i * 2];
            int col = i % 2, row = i / 2;
            double x0 = GAP + col * (PW + GAP), y0 = H - (GAP + row * (PH + HEAD + GAP)) - HEAD - PH;
            // brighten: read pixels, scale by max
            CGImageRef img = Load(png);
            std::vector<uint8_t> px(SW * SH * 4);
            CGContextRef tmp = CGBitmapContextCreate(px.data(), SW, SH, 8, SW * 4, cs, kCGImageAlphaPremultipliedLast);
            CGContextDrawImage(tmp, CGRectMake(0, 0, SW, SH), img);
            int mx = 1; for (size_t k = 0; k < px.size(); k += 4) mx = std::max({mx, (int)px[k], (int)px[k+1], (int)px[k+2]});
            double g = 230.0 / mx; long lit = 0;
            for (size_t k = 0; k < px.size(); k += 4) { bool on = px[k] | px[k+1] | px[k+2]; if (on) lit++; for (int c = 0; c < 3; c++) px[k+c] = (uint8_t)std::min(255.0, px[k+c] * g + (on ? 25 : 0)); }
            CGImageRef bright = CGBitmapContextCreateImage(tmp);
            // panel background = "no reflection written" (dark), image flipped upright
            CGContextSaveGState(ctx);
            CGContextTranslateCTM(ctx, x0, y0 + PH);
            CGContextScaleCTM(ctx, s, -s);
            CGContextDrawImage(ctx, CGRectMake(0, 0, SW, SH), bright);
            CGContextRestoreGState(ctx);
            // true pane footprint (kernel rows are bottom-up = CG y)
            CGContextSetRGBStrokeColor(ctx, .2, 1, .3, 1); CGContextSetLineWidth(ctx, 2.5);
            CGContextStrokeRect(ctx, CGRectMake(x0 + 573 * s, y0 + 285 * s, (905 - 573 + 1) * s, (534 - 285 + 1) * s));
            CGContextSetRGBStrokeColor(ctx, .35, .35, .38, 1); CGContextSetLineWidth(ctx, 1);
            CGContextStrokeRect(ctx, CGRectMake(x0, y0, PW, PH));
            Text(ctx, title, x0, y0 + PH + 24, 17, white);
            char sub[160]; snprintf(sub, sizeof sub, "%ld px written by the trace  ·  green = where the pane really is (83,250 px)", lit);
            Text(ctx, sub, x0, y0 + PH + 6, 13, grey);
            CGImageRelease(bright); CGContextRelease(tmp); CGImageRelease(img);
        }
        CGImageRef out = CGBitmapContextCreateImage(ctx);
        NSURL* url = [NSURL fileURLWithPath:[NSString stringWithUTF8String:argv[1]]];
        CGImageDestinationRef dst = CGImageDestinationCreateWithURL((__bridge CFURLRef)url, (__bridge CFStringRef)UTTypeJPEG.identifier, 1, nullptr);
        NSDictionary* opts = @{(id)kCGImageDestinationLossyCompressionQuality: @0.85};
        CGImageDestinationAddImage(dst, out, (__bridge CFDictionaryRef)opts);
        CGImageDestinationFinalize(dst);
        CFRelease(dst); CGImageRelease(out); CGContextRelease(ctx); CGColorSpaceRelease(cs);
        printf("wrote %s (%dx%d)\n", argv[1], W, H);
    }
    return 0;
}
